+def load_genres():
+    conn = get_db_connection()
+    cur = conn.cursor()
+    cur.execute("SELECT DISTINCT genres FROM movies WHERE genres IS NOT NULL AND genres != ''")
+    rows = cur.fetchall()
+    conn.close()
+    
+    genres_set = set()
+    for row in rows:
+        parts = [g.strip() for g in row['genres'].split(',')]
+        genres_set.update(parts)
+    return sorted(list(genres_set))

+def search_movies(query, genre_filter, status_filter, path_filter):
+    conn = get_db_connection()
+    cur = conn.cursor()
+    
+    sql = "SELECT id, title, year, watch_status, genres, imdb_rating, douban_rating FROM movies WHERE 1=1"
+    params = []
+    
+    if query:
+        sql += " AND (title LIKE ? OR original_title LIKE ? OR director LIKE ? OR actors LIKE ?)"
+        query_param = f"%{query}%"
+        params.extend([query_param, query_param, query_param, query_param])
+        
+    if genre_filter and genre_filter not in ["All", "全部"]:
+        sql += " AND genres LIKE ?"
+        params.append(f"%{genre_filter}%")
+        
+    if status_filter != "All":
+        sql += " AND watch_status = ?"
+        params.append("Seen" if status_filter == "Seen" else "Unseen")
+        
+    if path_filter == "With File Path":
+        sql += " AND physical_path IS NOT NULL AND physical_path != ''"
+    elif path_filter == "Without File Path":
+        sql += " AND (physical_path IS NULL OR physical_path = '')"
+        
+    sql += " ORDER BY id DESC"
+    
+    cur.execute(sql, params)
+    rows = cur.fetchall()
+    conn.close()
+    return rows

+def save_env_vars(vars_dict):

+    current_vars = {}
+    if os.path.exists(env_path):

+                        parts = line.split('=', 1)

+                            current_vars[parts[0].strip()] = parts[1].strip()
+        except Exception:
+            pass
+    
+    current_vars.update(vars_dict)
+    
+    try:
+        with open(env_path, 'w') as f:
+            for k, v in current_vars.items():

+    except Exception:
